# Team Pulse — Cron Schedule
#
# 7:00 AM Bangkok (00:00 UTC) — Morning pipeline (extract AM snapshot, compile, morning briefing)
# 6:00 PM Bangkok (11:00 UTC) — Evening pipeline (extract PM snapshot, compile, EOD summary)
#
# Install with: bash scripts/setup-cron.sh
# Or manually: crontab scripts/crontab.txt
#
# Prerequisites:
#   - .env file configured with ANTHROPIC_API_KEY
#   - gh CLI authenticated
#   - python3 available at the path below

# Adjust PROJECT_DIR to your installation path
PROJECT_DIR=/home/tkhongsap/my-github/team-pulse-example
SHELL=/bin/bash
PATH=/usr/local/bin:/usr/bin:/bin

# 7:00 AM Bangkok = 00:00 UTC — Morning pipeline
0 0 * * * cd $PROJECT_DIR && python3 scripts/run_pipeline.py --period am >> scripts/pipeline.log 2>&1

# 6:00 PM Bangkok = 11:00 UTC — Evening pipeline
0 11 * * * cd $PROJECT_DIR && python3 scripts/run_pipeline.py --period pm >> scripts/pipeline.log 2>&1
